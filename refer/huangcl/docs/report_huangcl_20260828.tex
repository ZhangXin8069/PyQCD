% F05：物理核心公式
\begin{frame}[t]{OPE 的物理核心是“局域场强—平行移动—非局域迹”的规范协变组合}
  \begin{columns}[T,totalwidth=\textwidth]
    \begin{column}{0.48\textwidth}
      \begin{align*}
        F_{\mu\nu}^{\rm clov}(x)
        &\propto -\frac{i}{8}\sum_{q=1}^{4}
        \left(P^{(q)}_{\mu\nu}(x)-P^{(q)\dagger}_{\mu\nu}(x)\right),\\
        \widetilde F_{\mu\nu}(x)
        &=\frac12\epsilon_{\mu\nu\rho\sigma}F_{\rho\sigma}(x),\\
        M^{\mu\nu;\rho\sigma}(z)
        &=\sum_{\boldsymbol x}\operatorname{Tr}\!\Bigl[
        F_{\mu\nu}(x+z\hat{\ell})W_z^\dagger\\
        &\qquad\widetilde F_{\rho\sigma}(x)W_z\Bigr],\\
        O_{\rm code}(z)
        &=-M^{30;30}(z)-M^{31;31}(z)\\
        &\quad+2M^{01;01}(z).
      \end{align*}
      \par\vspace{0em}\footnotesize\textbf{检查：}反厄米化保证共轭；$W_z$ 连接两端；$z=0$ 时 $W_z=1$；格点归一化待重整化补齐。
    \end{column}
    \begin{column}{0.41\textwidth}
      \small
      \begin{tabularx}{0.98\linewidth}{@{}>{\raggedright\arraybackslash}X@{}}
        $\text{Input: }U_\mu(x)\in SU(3),\quad \ell=\texttt{link\_dir}$ \\
        $\text{Build: 四个平移 plaquette}\to F_{\mu\nu}^{\rm clov}$ \\
        $\text{Dual: }\widetilde F=\tfrac12\epsilon F\text{（代码 Tensor4）}$ \\
        $\text{Transport: }F(x+z\hat\ell)\xrightarrow{W_z^\dagger}\widetilde F(x)\xrightarrow{W_z}$ \\
        $\text{Trace: }\operatorname{Tr}_{c}\text{ 与空间求和}$ \\
        $\text{Output: }O(t,z)\in\mathbb C^{N_t\times\Delta z}$

% F09：统计与比值伪代码
\begin{frame}[t]{比值步骤把“真空扣除”放在重采样之后，避免把相关性误当成物理信号}
  \centering
  \small
  \begin{tabular}{@{}l@{}}
    $\text{Input: }C_2^{(k)}(t_{\rm sink},t_{\rm src}),\ O^{(k)}(\tau,z),\quad k=1,\ldots,N_{\rm conf}$ \\

    $\text{Construct: }C_3^{(k)}(t,\tau,z)=O^{(k)}(\tau,z)C_2^{(k)}(t)$ \\

% F14：FH 算法
\begin{frame}[t]{FH 变换把插入时间求和的斜率信息转为更适合平台拟合的差分量}
  \begin{columns}[T,totalwidth=\textwidth]
    \begin{column}{0.53\textwidth}
      \begin{align*}
        T_n(t,z)&=\sum_{\tau=n}^{t-n}R(t,\tau,z),\\
        FH_n(t,z)&=T_n(t+1,z)-T_n(t,z)\\
        &=\sum_{\tau=n}^{t+1-n}R(t+1,\tau,z)
        -\sum_{\tau=n}^{t-n}R(t,\tau,z).
      \end{align*}
      \begin{block}{边界与模型}
        $t\le 2n$ 的点无意义；代码从 $2n$ 后计算。$P=4$ 使用 $n=2$、$t=7\!:\!10$ 的常数模型 $FH(t)=c_0$。
      \end{block}
    \end{column}
    \begin{column}{0.43\textwidth}
      \small
      \begin{tabular}{@{}l@{}}
        $\text{Input: }R\in\mathbb R^{N_{\rm sample}\times t\times\tau\times z}$ \\
        $\text{Average: }R\text{ 的六个方向 }\pm x,\pm y,\pm z$ \\
        $\text{Trim: }\tau=n,\ldots,t-n$ \\
        $\text{Sum: }T_n(t)=\sum_\tau R(t,\tau)$ \\
        $\text{Difference: }FH_n(t)=T_n(t)-T_n(t-1)$ \\
        $\text{Fit: }FH_n(t,z)\to c_0(z)$ \\
        $\text{Output: }FH\text{ 数组、误差条、}c_0(z)$
      \end{tabular}
    \end{column}
  \end{columns}
  \source{实现：05\_FH\_bare\_matele/1\_preprocess\_FH.py:125--156；一体脚本:216--288。}
\end{frame}
